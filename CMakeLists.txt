cmake_minimum_required(VERSION 3.22)

project(safeMRC_for_DM_firmware LANGUAGES C ASM)

# The CubeMX toolchain file supplies a generic linker command. The project
# keeps the linker layout in cmake/STM32H723VGHx.ld so DMA buffers and stack
# placement stay under version control.
set(CMAKE_C_LINK_FLAGS "")
set(CMAKE_CXX_LINK_FLAGS "")

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

set(MRC_COM_BACKEND "RS485" CACHE STRING "Communication backend (RS485 or CAN)")
set_property(CACHE MRC_COM_BACKEND PROPERTY STRINGS RS485 CAN)
string(TOUPPER "${MRC_COM_BACKEND}" MRC_COM_BACKEND)
if(MRC_COM_BACKEND STREQUAL "RS485")
    set(MRC_COM_BACKEND_VALUE 0)
elseif(MRC_COM_BACKEND STREQUAL "CAN")
    set(MRC_COM_BACKEND_VALUE 1)
else()
    message(FATAL_ERROR "MRC_COM_BACKEND must be RS485 or CAN, got '${MRC_COM_BACKEND}'")
endif()

# Keep the source manifest for the application layer. CubeMX owns the Core/
# and HAL source lists in cmake/stm32cubemx/CMakeLists.txt.
include(cmake/stm32h723_sources.cmake)

add_executable(${PROJECT_NAME})

# This file is generated by CubeMX and should remain the source of truth for
# Core/, Drivers/, include paths, and MCU feature macros.
add_subdirectory(cmake/stm32cubemx)

# CubeMX currently emits a startup file that calls ExitRun0Mode(), which is not
# part of this application. Replace only that generated source with the
# maintained startup file; all other generated sources remain in the target.
get_target_property(MX_SOURCES ${PROJECT_NAME} SOURCES)
list(FILTER MX_SOURCES EXCLUDE REGEX "/startup_stm32h723xx\\.s$")
list(FILTER MX_SOURCES EXCLUDE REGEX "/sysmem\\.c$")
set_property(TARGET ${PROJECT_NAME} PROPERTY SOURCES "${MX_SOURCES}")

target_sources(${PROJECT_NAME} PRIVATE
    ${FIRMWARE_COMMON_SOURCES}
    ${FIRMWARE_DEVICE_SOURCES}
    Common/Src/sysmem.c
    startup/startup_stm32h723xx_gcc.s
)

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Common/Inc
    ${CMAKE_CURRENT_SOURCE_DIR}/Devices/Inc
)

target_compile_definitions(${PROJECT_NAME} PRIVATE
    MRC_COM_BACKEND=${MRC_COM_BACKEND_VALUE}
)

target_compile_options(${PROJECT_NAME} PRIVATE
    -fno-common
    $<$<CONFIG:Debug>:-Og;-g3>
    $<$<CONFIG:Release>:-O2>
)

# Keep the custom memory layout: DMA buffers marked RAM_D2 must remain in D2
# SRAM, while normal data uses the available AXI SRAM.
target_link_options(${PROJECT_NAME} PRIVATE
    -T${CMAKE_CURRENT_SOURCE_DIR}/cmake/STM32H723VGHx.ld
    --specs=nosys.specs
    -u _printf_float
    -Wl,--gc-sections
    -Wl,--print-memory-usage
    -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/safeMRC_for_DM_firmware.map
)
set_target_properties(${PROJECT_NAME} PROPERTIES
    OUTPUT_NAME safeMRC_for_DM_firmware
    SUFFIX .elf
)

add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O ihex
            $<TARGET_FILE:${PROJECT_NAME}>
            ${CMAKE_CURRENT_BINARY_DIR}/safeMRC_for_DM_firmware.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary
            $<TARGET_FILE:${PROJECT_NAME}>
            ${CMAKE_CURRENT_BINARY_DIR}/safeMRC_for_DM_firmware.bin
    VERBATIM
)

add_custom_target(size
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${PROJECT_NAME}>
    DEPENDS ${PROJECT_NAME}
    COMMENT "Firmware size (map: ${CMAKE_CURRENT_BINARY_DIR}/safeMRC_for_DM_firmware.map)"
    VERBATIM
)

# Flash targets remain optional and do not make a connected probe necessary
# for ordinary compilation.
add_custom_target(flash
    COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/scripts/flash_openocd.sh
            $<TARGET_FILE:${PROJECT_NAME}>
    DEPENDS ${PROJECT_NAME}
    COMMENT "Flash and verify firmware with OpenOCD"
    USES_TERMINAL
    VERBATIM
)

add_custom_target(flash-stlink
    COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/scripts/flash_stlink.sh
            ${CMAKE_CURRENT_BINARY_DIR}/safeMRC_for_DM_firmware.bin
    DEPENDS ${PROJECT_NAME}
    COMMENT "Flash firmware with st-flash"
    USES_TERMINAL
    VERBATIM
)
